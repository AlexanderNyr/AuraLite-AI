# AuraLite app's build (2.21): exactly what packaging/app/build_app.py freezes and what

#
#     pip install -r packaging/app/requirements.txt
#     pip install --no-deps .

# build_app.py refuses to build when what is installed differs from these pins.

# Qt 6.9.3: the last Qt for processors without SSE4.2 / POPCNT, and the oldest the
# `studio` extra allows (its Visual C++ runtime, 14.44, is what PyTorch 2.9+ needs).

shiboken6==6.9.3

# pictures the Images tab's generator trains on and measures its quality.
safetensors==0.8.0
tokenizers==0.23.3
zstandard==0.25.0
psutil==7.2.2
pillow==13.0.0
# The freezer.
pyinstaller==6.22.3
